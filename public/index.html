async function connectWaveChannel() { if (waveRoom && waveRoom.state === "connected") return true; if (waveRoomPromise) return waveRoomPromise; waveRoomPromise = (async () => { try { if (!window.LivekitClient || !currentUser.radCode) return false; if (waveRoom) { await waveRoom.disconnect(); waveRoom = null; } const tokenRes = await fetch(SERVER_URL + '/api/livekit/token', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ roomId: currentUser.waveId, deviceId: currentUser.id, userName: currentUser.name }) }); const tokenData = await tokenRes.json(); // ✅ التعديل الجديد: التأكد من وجود توكن ورابط if (!tokenData.token) { throw new Error("السيرفر لم يرجع تصريح الدخول (Token)"); } // ✅ أخذ رابط LiveKit من السيرفر مباشرة، وإذا لم يوجد نستخدم المشفر let lkUrl = tokenData.url || LIVEKIT_URL; // ✅ إصلاح تلقائي: LiveKit يتطلب wss وليس https if (lkUrl.startsWith('https://')) lkUrl = lkUrl.replace('https://', 'wss://'); if (lkUrl.startsWith('http://')) lkUrl = lkUrl.replace('http://', 'ws://'); console.log("Connecting to LiveKit URL:", lkUrl); // ستراها في الكونسول waveRoom = new window.LivekitClient.Room(); waveRoom.on(window.LivekitClient.RoomEvent.Connected, () => { updateWaveUIStatus('connected'); }); waveRoom.on(window.LivekitClient.RoomEvent.TrackSubscribed, (track, pub, participant) => { if (track.kind === "audio") { const audioElement = track.attach(); audioElement.volume = 1.0; document.body.appendChild(audioElement); const playPromise = audioElement.play(); if (playPromise !== undefined) { playPromise.catch(error => { console.warn("Autoplay prevented:", error); toast("⚠️ يرجى النقر على الشاشة لتفعيل الصوت", "error"); document.addEventListener('click', () => { audioElement.play().catch(()=>{}); }, { once: true }); }); } } }); waveRoom.on(window.LivekitClient.RoomEvent.TrackUnsubscribed, (track) => { track.detach().forEach(el => el.remove()); }); waveRoom.on(window.LivekitClient.RoomEvent.Disconnected, () => { waveRoom = null; if (isTalking) stopTalking(); updateWaveUIStatus('disconnected'); }); await waveRoom.connect(lkUrl, tokenData.token); return true; } catch (e) { console.error("Wave Connect Error:", e); // ✅ إظهار الخطأ الحقيقي للمستخدم toast("⚠️ " + (e.message || "فشل الاتصال بالخادم الصوتي"), "error"); waveRoom = null; return false; } finally { waveRoomPromise = null; } })(); return waveRoomPromise; }
